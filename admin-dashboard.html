/**************************************************** * ADMIN DASHBOARD DATA * Reads students directly from STUDENTS sheet ****************************************************/ function getAdminDashboardData() { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const studentsSheet = ss.getSheetByName("STUDENTS"); const facultySheet = ss.getSheetByName("FACULTY"); let students = 0; let faculty = 0; /************************************************ * STUDENTS ************************************************/ if (studentsSheet) { const data = studentsSheet.getDataRange().getValues(); if (data.length > 1) { const headers = data[0]; const statusIndex = headers.indexOf("Status"); for (let i = 1; i < data.length; i++) { // Ignore completely empty rows if ( data[i].every(function(value) { return String(value).trim() === ""; }) ) { continue; } // If Status column exists if (statusIndex !== -1) { const status = String(data[i][statusIndex]) .trim() .toLowerCase(); if ( status === "active" || status === "" ) { students++; } } else { // Count student if no Status column students++; } } } } /************************************************ * FACULTY ************************************************/ if (facultySheet) { const data = facultySheet.getDataRange().getValues(); if (data.length > 1) { for (let i = 1; i < data.length; i++) { if ( String(data[i][0]).trim() !== "" ) { faculty++; } } } } /************************************************ * RETURN ************************************************/ return { success: true, students: students, faculty: faculty, attendance: 0, fees: 0 }; } catch (error) { return { success: false, error: error.toString() }; } }
